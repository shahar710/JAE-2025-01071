# --- 2. GENERATE HULLS (The Base R Way) ---
# We use chull() to find points and store them in a standard dataframe
hulls_df <- data.frame()

for(g in levels(hull_data$group_factor)) {
  sub_dat <- hull_data %>% filter(group_factor == g)
  
  # Calculate hull indices
  if(nrow(sub_dat) >= 3) {
    hull_idx <- chull(sub_dat$PCoA1, sub_dat$PCoA2)
    
    # Extract points and close the loop (first point = last point)
    hull_pts <- sub_dat[c(hull_idx, hull_idx[1]), ]
    
    # Add to master dataframe
    hulls_df <- rbind(hulls_df, hull_pts)
  }
}

# --- 3. BACKGROUND STATS (Using SF for intersection math only) ---
overlap_results <- data.frame()
# Helper to make SF polygon from points
make_poly <- function(df) {
  mat <- as.matrix(df[, c("PCoA1", "PCoA2")])
  st_polygon(list(mat)) %>% st_sfc() %>% st_as_sf()
}

# Loop for overlaps (Background calc)
groups <- levels(hull_data$group_factor)
pairs <- combn(groups, 2, simplify = FALSE)

for(p in pairs) {
  g1 <- p[1]; g2 <- p[2]
  df1 <- hulls_df %>% filter(group_factor == g1)
  df2 <- hulls_df %>% filter(group_factor == g2)
  
  if(nrow(df1) > 0 && nrow(df2) > 0) {
    poly1 <- make_poly(df1); poly2 <- make_poly(df2)
    a1 <- st_area(poly1); a2 <- st_area(poly2)
    inter <- st_intersection(poly1, poly2)
    a_shared <- if(length(inter) > 0) st_area(inter) else 0
    
    overlap_results <- rbind(overlap_results, data.frame(
      Group_A = g1, Group_B = g2,
      Overlap_Pct_of_A = round(as.numeric(a_shared/a1)*100, 1),
      Overlap_Pct_of_B = round(as.numeric(a_shared/a2)*100, 1)
    ))
  }
}

    delta_1 = Med_PCoA1 - RS_PCoA1, # Change in PCoA1
    delta_2 = Med_PCoA2 - RS_PCoA2  # Change in PCoA2
  ) %>%
  select(species, delta_1, delta_2)

# 2. Get Trait Vectors (The "Environment Arrows")
# We use arrow_df which contains the PCoA1/PCoA2 coordinates of each trait tip
trait_vectors <- arrow_df %>%
  select(trait, Axis.1, Axis.2)

# 3. Calculate Projections (The "Alignment")
# We loop through every species and every trait to see how well they align
projections <- expand.grid(species = species_vectors$species, 
                           trait = trait_vectors$trait) %>%
  left_join(species_vectors, by = "species") %>%
  left_join(trait_vectors, by = "trait") %>%
  # --- THE CORE CALCULATION (Dot Product) ---
  mutate(
    # How much of the species' shift is explained by this trait?
    projection = (delta_1 * Axis.1) + (delta_2 * Axis.2)
  ) %>%
  group_by(species) %>%
  # Normalize so we can compare (optional, but helps interpretation)
  arrange(species, desc(abs(projection)))

# 4. View the "Main Driver" for each species
top_contributors <- projections %>%
  slice(1) %>% # Get the #1 strongest driver per species
  select(species, trait, projection) %>%
  mutate(direction = ifelse(projection > 0, "Increased", "Decreased"))

  st_sf(geometry = st_sfc(st_polygon(list(poly_coords))))
